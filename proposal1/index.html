timescaledb_continuous_agg.sql
POSTGRESQL + TIMESCALEDB
-- Hypertable creation with 1-hour chunk interval for high-frequency tick capture
CREATE TABLE market_ticks (
    recorded_at TIMESTAMPTZ NOT NULL,
    symbol      TEXT NOT NULL,
    price       NUMERIC(12, 4) NOT NULL,
    volume      INTEGER NOT NULL,
    bid_depth   NUMERIC(12, 2),
    ask_depth   NUMERIC(12, 2)
);
SELECT create_hypertable('market_ticks', 'recorded_at', chunk_time_interval => INTERVAL '1 hour');

-- Continuous Aggregate for Real-Time 1-Minute OHLCV & Volume Weighted Average Price (VWAP)
CREATE MATERIALIZED VIEW ohlcv_1m
WITH (timescaledb.continuous) AS
SELECT
    time_bucket('1 minute', recorded_at) AS bucket,
    symbol,
    first(price, recorded_at)            AS open,
    max(price)                           AS high,
    min(price)                           AS low,
    last(price, recorded_at)             AS close,
    sum(volume)                          AS total_volume,
    sum(price * volume) / sum(volume)    AS vwap
FROM market_ticks
GROUP BY bucket, symbol;

-- Refresh policy: real-time updates every 10 seconds
SELECT add_continuous_aggregate_policy('ohlcv_1m',
    start_offset => INTERVAL '2 hours',
    end_offset   => INTERVAL '10 seconds',
    schedule_interval => INTERVAL '10 seconds');
